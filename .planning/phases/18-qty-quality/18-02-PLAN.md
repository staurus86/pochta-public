  <action>
    STEP 1 — Obtain admin token:

      echo "Token: $TOKEN"

    STEP 2 — Trigger reanalysis for project-3:
      curl -s -X POST https://pochta-production.up.railway.app/api/projects/project-3-mailbox-file/reanalyze \
        -H "Authorization: Bearer $TOKEN" | python -m json.tool

      Wait for completion — poll GET /api/projects/project-3-mailbox-file/status or wait 90 seconds.
      Log the number of messages reanalyzed.

    STEP 3 — Trigger reanalysis for project-4:
      curl -s -X POST https://pochta-production.up.railway.app/api/projects/project-4-klvrt-mail/reanalyze \
        -H "Authorization: Bearer $TOKEN" | python -m json.tool

      Wait for completion — poll or wait 90 seconds.
      Log the number of messages reanalyzed.

    STEP 4 — Capture baseline_v9:
      python scripts/audit_baseline.py \
        --token "$TOKEN" \
        --limit 300 \
        --seed 42 \
        --skip-n8n \
        --out scripts/baselines/baseline_v9.json

    STEP 5 — Print the qty section from baseline_v9:
      python -c "
      import json
      with open('scripts/baselines/baseline_v9.json') as f:
          b = json.load(f)
      q = b.get('fields', {}).get('qty', {})
      print('qty.present:   ', q.get('present_pct', '?'))
      print('qty.noise_free:', q.get('noise_free_pct', '?'))
      print('qty.noise:     ', q.get('noise_pct', '?'))
      print('baseline_v8 qty.noise_free was: 4.0%')
      "

    STEP 6 — Compare to baseline_v8:
      python -c "
      import json
      with open('scripts/baselines/baseline_v8.json') as f:
          v8 = json.load(f)
      with open('scripts/baselines/baseline_v9.json') as f:
          v9 = json.load(f)
      q8 = v8.get('fields', {}).get('qty', {})
      q9 = v9.get('fields', {}).get('qty', {})
      print('=== qty delta baseline_v8 -> baseline_v9 ===')
      for k in ('present_pct', 'noise_free_pct', 'noise_pct'):
          a = q8.get(k, 'N/A')
          b = q9.get(k, 'N/A')
          print(f'  {k}: {a} -> {b}')
      target = 30.0
      nf = q9.get('noise_free_pct', 0)
      result = 'PASS' if nf >= target else 'FAIL (shortfall: ' + str(round(target - nf, 1)) + 'pp)'
      print()
      print(f'Target >=30%: {result}')
      "

    STEP 7 — If noise_free_pct < 30%, print diagnostic:
      python -c "
      import json
      with open('scripts/baselines/baseline_v9.json') as f:
          b = json.load(f)
      n = b.get('fields', {}).get('qty', {}).get('noise_count', 0)
      p = b.get('fields', {}).get('qty', {}).get('noise_free_pct', 0)
      print(f'noise_count={n}, noise_free_pct={p}%')
      print('Expected: noise_free_pct >= 30% (estimate was 40-46%)')
      "
      Report the shortfall clearly so the user can assess whether further investigation is needed.

    STEP 8 — Commit baseline_v9:
      git add scripts/baselines/baseline_v9.json
      git commit -m "docs(18): capture baseline_v9 (qty noise_free v2 metric)"
